rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function isTeacher() {
      return signedIn() && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }
    function ownsEvent(eid) {
      return signedIn() &&
        get(/databases/$(database)/documents/events/$(eid)).data.owner == request.auth.uid;
    }

    // 체험학습: 만든 선생님만 보고 고칠 수 있음 (학생 익명 계정은 만들 수 없음)
    match /events/{eid} {
      allow read, delete: if signedIn() && resource.data.owner == request.auth.uid;
      allow create: if isTeacher() && request.resource.data.owner == request.auth.uid;
      allow update: if signedIn() && resource.data.owner == request.auth.uid
                    && request.resource.data.owner == resource.data.owner;

      // 설문: 학생도 읽을 수 있음(질문만 들어 있음), 고치는 건 선생님만
      match /surveys/{sid} {
        allow read: if signedIn();
        allow create, update: if ownsEvent(eid) && request.resource.data.owner == request.auth.uid;
        allow delete: if ownsEvent(eid);

        // 응답: 기기(익명 계정)마다 한 번만, 설문이 열려 있을 때만
        match /responses/{rid} {
          allow create: if signedIn() && request.auth.uid == rid
            && get(/databases/$(database)/documents/events/$(eid)/surveys/$(sid)).data.open == true
            && request.resource.data.keys().hasOnly(['answers', 'at'])
            && request.resource.data.answers is map
            && request.resource.data.answers.size() <= 30;
          allow read: if signedIn() && (request.auth.uid == rid || ownsEvent(eid));
          allow delete: if ownsEvent(eid);
        }
      }

      // 준비물 목록: 학생도 읽을 수 있음(목록만 들어 있음), 고치는 건 선생님만
      match /lists/{lid} {
        allow read: if signedIn();
        allow create, update: if ownsEvent(eid) && request.resource.data.owner == request.auth.uid;
        allow delete: if ownsEvent(eid);

        // "다 챙겼어요" 알림: 기기(익명 계정)마다 하나, 시간만 저장
        match /checks/{rid} {
          allow create: if signedIn() && request.auth.uid == rid
            && request.resource.data.keys().hasOnly(['at']);
          allow read: if signedIn() && (request.auth.uid == rid || ownsEvent(eid));
          allow delete: if ownsEvent(eid);
        }
      }

      // 안전 교육: 학생도 읽을 수 있음(영상 주소·안내만), 고치는 건 선생님만
      match /safety/{sid} {
        allow read: if signedIn();
        allow create, update: if ownsEvent(eid) && request.resource.data.owner == request.auth.uid;
        allow delete: if ownsEvent(eid);

        // 시청 체크: 번호마다 하나, 번호와 시청 장소만 저장 (이름 없음)
        match /views/{n} {
          allow read: if signedIn();
          allow create: if signedIn()
            && request.resource.data.keys().hasOnly(['n', 'place', 'by', 'at'])
            && request.resource.data.by == request.auth.uid
            && request.resource.data.n is int
            && string(request.resource.data.n) == n
            && request.resource.data.n >= 1
            && request.resource.data.n <= get(/databases/$(database)/documents/events/$(eid)/safety/$(sid)).data.total
            && request.resource.data.place in ['home', 'class', 'alone', 'teacher']
            && (ownsEvent(eid) || get(/databases/$(database)/documents/events/$(eid)/safety/$(sid)).data.open == true);
          allow delete: if ownsEvent(eid);
        }
      }
    }
  }
}
